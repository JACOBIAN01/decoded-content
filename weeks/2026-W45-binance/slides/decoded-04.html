The fix, in code
// BUG: obey any trusted machine
if (req.from.trusted) sign(req);

// FIX: one hacked box isn't enough
if (req.amount > CAP && req.ok < 2)
  wait();   // needs a 2nd human
else if (hourly(req.from) > LIMIT)
  block();  // speed limit
else sign(req);
